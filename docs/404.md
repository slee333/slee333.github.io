<script>
  document.addEventListener("DOMContentLoaded", function() {
    const path = window.location.pathname;
    const langData = {{ site.data.404 | jsonify }};
    const defaultLang = "ko"; 
    let langToShow = defaultLang;

    for (const data of langData) {
      if (path.startsWith(`/${data.lang}/`)) {
        langToShow = data.lang;
        break;
      }
    }
    
    // 만약 ko 언어인데 경로가 /ko/로 시작하지 않는 경우 처리
    if (langToShow === defaultLang && !path.startsWith(`/${defaultLang}/`)) {
      const isAnyLangPath = langData.some(data => path.startsWith(`/${data.lang}/`));
      if (!isAnyLangPath) {
        langToShow = defaultLang;
      }
    }

    const contentEl = document.getElementById(`lang-${langToShow}`);
    if (contentEl) {
      contentEl.style.display = 'block';
    } else {
      document.getElementById(`lang-${defaultLang}`).style.display = 'block';
    }

    // 브랜드 로고 링크를 언어에 맞게 동적으로 변경
    const brandLink = document.getElementById('brand-link');
    const baseUrl = "{{ '/' | relative_url }}";

    if (brandLink) {
      if (langToShow === defaultLang || langToShow === "") {
        brandLink.href = baseUrl;
      } else {
        // baseurl이 / 일때와 아닐때를 모두 고려
        const newHref = baseUrl.endsWith('/') ? `${baseUrl}${langToShow}/` : `${baseUrl}/${langToShow}/`;
        brandLink.href = newHref;
      }
    }
  });
</script>
